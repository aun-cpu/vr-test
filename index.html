AFRAME.registerComponent('tap-to-action', { init: function () { const el = this.el; let mixer = null; let actions = {}; let activeAction = null; let isActionPlaying = false; // モデルロード完了時に Three.js のミキサーとクリップを準備 el.addEventListener('model-loaded', () => { const model = el.getObject3D('mesh'); if (!model || !model.animations || model.animations.length === 0) return; mixer = new THREE.AnimationMixer(model); model.animations.forEach((clip) => { actions[clip.name] = mixer.clipAction(clip); }); if (actions['idle']) { activeAction = actions['idle']; activeAction.play(); } if (actions['action']) { actions['action'].setLoop(THREE.LoopOnce, 1); actions['action'].clampWhenFinished = true; } // アニメーション終了検知 mixer.addEventListener('finished', (e) => { if (e.action === actions['action'] && isActionPlaying) { isActionPlaying = false; console.log("action 終了: 位置とモーションを滑らかに戻します"); // ★ 1. 位置を 0.4 秒かけて滑らかに「0 0 -0.5」へ戻す el.removeAttribute('animation__move'); el.setAttribute('animation__move', { property: 'position', to: '0 0 -0.5', dur: 400, easing: 'easeInOutQuad' }); // ★ 2. action から idle へ 0.4 秒かけて滑らかにクロスフェード const actionClip = actions['action']; const idleClip = actions['idle']; idleClip.reset(); idleClip.play(); actionClip.crossFadeTo(idleClip, 0.4, true); activeAction = idleClip; } }); }); // 毎フレームミキサーを更新 this.tick = function (t, dt) { if (mixer) mixer.update(dt / 1000); }; // 画面タップイベント window.addEventListener('click', () => { if (!isActionPlaying && actions['action'] && actions['idle']) { isActionPlaying = true; console.log("action を即座に再生"); // ★ タップ時は即座に上がった位置へ動かす（元の動作を維持） el.removeAttribute('animation__move'); el.setAttribute('position', '0 0.02 -0.5'); actions['idle'].stop(); actions['action'].reset(); actions['action'].play(); activeAction = actions['action']; } }); } });
